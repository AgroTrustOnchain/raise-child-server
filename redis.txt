package main

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"time"

	"://github.com"
	"://github.com"
)

var (
	ctx = context.Background()
	rdb *redis.Client
)

// ResponseRecorder dùng để bắt lấy nội dung mà Controller ghi ra
type bodyLogWriter struct {
	gin.ResponseWriter
	body *bytes.Buffer
}

func (w bodyLogWriter) Write(b []byte) (int, error) {
	w.body.Write(b)
	return w.ResponseWriter.Write(b)
}

// RedisCacheMiddleware - Middleware xử lý cache
func RedisCacheMiddleware(duration time.Duration) gin.HandlerFunc {
	return func(c *gin.Context) {
		// Chỉ cache các request GET
		if c.Request.Method != http.MethodGet {
			c.Next()
			return
		}

		// 1. Tạo Key dựa trên URL (bao gồm cả Query Params)
		cacheKey := fmt.Sprintf("cache:%s", c.Request.RequestURI)

		// 2. Kiểm tra Redis
		val, err := rdb.Get(ctx, cacheKey).Result()
		if err == nil {
			// Cache HIT: Trả về dữ liệu luôn
			c.Header("Content-Type", "application/json")
			c.Header("X-Cache", "HIT")
			c.String(http.StatusOK, val)
			c.Abort() // Dừng xử lý các handler tiếp theo
			return
		}

		// 3. Cache MISS: Ghi đè ResponseWriter để lấy kết quả sau khi Controller chạy xong
		blw := &bodyLogWriter{body: bytes.NewBufferString(""), ResponseWriter: c.Writer}
		c.Writer = blw

		c.Next() // Cho phép request đi tiếp vào Controller

		// 4. Lưu kết quả vào Redis nếu status code là 200 OK
		if c.Writer.Status() == http.StatusOK {
			rdb.Set(ctx, cacheKey, blw.body.String(), duration)
		}
	}
}

func main() {
	// Khởi tạo Redis
	rdb = redis.NewClient(&redis.Options{Addr: "localhost:6379"})

	r := gin.Default()

	// Sử dụng middleware cho một group hoặc toàn bộ router
	r.GET("/products", RedisCacheMiddleware(5*time.Minute), func(c *gin.Context) {
		// Giả lập xử lý nặng từ Database
		time.Sleep(2 * time.Second) 
		c.JSON(http.StatusOK, gin.H{
			"data": "Danh sách sản phẩm",
			"time": time.Now().Format(time.RFC3339),
		})
	})

	r.Run(":8080")
}
